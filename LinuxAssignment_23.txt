#!/bin/bash

# Configuration
SERVER_NAME="apache"
BACKUP_DIR="/backups"
DATE=$(date +%Y-%m-%d)
FILENAME="${SERVER_NAME}_backup_${DATE}.tar.gz"
LOG_FILE="${BACKUP_DIR}/backup_verification.log"

# Target directories to back up (Adjust /etc/httpd to /etc/apache2 if on Ubuntu)
TARGETS="/etc/httpd /var/www/html"

echo "=== Backup Started: $(date) ===" >> $LOG_FILE

# 1. Create Compressed Backup
tar -czf ${BACKUP_DIR}/${FILENAME} $TARGETS 2>> $LOG_FILE

# 2. Verify Backup Integrity
if [ -f "${BACKUP_DIR}/${FILENAME}" ]; then
    echo "SUCCESS: Backup created at ${BACKUP_DIR}/${FILENAME}" >> $LOG_FILE
    echo "--- Integrity Verification (File Contents) ---" >> $LOG_FILE
    tar -tzf ${BACKUP_DIR}/${FILENAME} >> $LOG_FILE
    echo "Integrity Check Passed for ${FILENAME}" >> $LOG_FILE
else
    echo "ERROR: Backup creation failed for ${FILENAME}" >> $LOG_FILE
fi

echo -e "=== Backup Ended ===\n" >> $LOG_FILE

##############################################################
# Add execution Permission to the script
chmod +x /home/sarah/apache_backup.sh
